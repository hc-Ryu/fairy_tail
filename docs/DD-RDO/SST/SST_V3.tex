\end{aligned}

\begin{itemize}
    \item scaling
    \begin{equation*}
        \begin{aligned}
        &f_X(x;d) = \left|\mathbf{J}\right|\cdot f_Z(z;g)\\
        \\
        &f_X(x;d)=\frac{1}{d}\cdot f_Z(z;g)\\
        \\
        &\log\:f_X(x;d)= - \log\:d + \log\:f_Z(z:g)\\
        \\
        &\frac{\partial}{\partial d}\:\log\:f_X(x;d) = -\frac{1}{d} + \frac{\partial}{\partial d}\:\log\:f_Z(z;g)\\
        \\
        &\hspace{80pt}=-\frac{1}{d}+\frac{\partial z}{\partial d}\cdot\frac{\partial}{\partial z}\cdot\:\log\:f_Z(z;g)\\
        \\
        &\hspace{80pt}=-\frac{1}{d}-\frac{x}{d^2}\cdot\textcolor{red}{\frac{\partial}{\partial z}\cdot\:\log\:f_Z(z;g)}\\
        \\
        &\hspace{80pt}=-\frac{1}{d}\left(1+\frac{x}{d}\cdot\textcolor{red}{\frac{\partial}{\partial z}\cdot\:\log\:f_Z(z;g)}\right)\\
        \\
        &\mathrm{S}_{d}(x;d) = -\frac{1}{d}\left(1+\frac{x}{d}\cdot\textcolor{red}{\mathrm{S}_{z}(z;g)}\right)\\
        \end{aligned}
    \end{equation*}
\end{itemize}
\begin{itemize}
    \item shifting
    \begin{equation*}
        \mathrm{S}_{d}(X;d) = -\mathrm{S}_{z}(Z;g)
    \end{equation*}
\end{itemize}

\begin{itemize}
    \item scaling
    \begin{equation}
        \begin{aligned}
        &f_{\mathbf{X}}(\mathbf{x};\mathbf{d}) = \frac{1}{\Pi_{i=1}^{10}d_i}\cdot f_{\mathbf{Z}}(\mathbf{z};\mathbf{g}) \\
        \\
        &\log\:f_{\mathbf{X}}(\mathbf{x};\mathbf{d})=-\sum_{i=1}^{10} \log\:d_i + \log\:f_{\mathbf{Z}}(\mathbf{z};\mathbf{g}) \\
        \\
        &\frac{\partial}{\partial d_{k}} \log f_{\mathbf{X}}(\mathbf{x};\mathbf{d}) = \frac{\partial}{\partial d_{k}} \left( -\sum_{i=1}^{10} \log d_i \right) + \frac{\partial}{\partial d_{k}} \log f_{\mathbf{Z}}(\mathbf{z};\mathbf{g})\\
        \\
        &\text{1st. term; Jacobian Term}\\
        \\
        &\frac{\partial}{\partial d_{k}} \left( -\sum_{i=1}^{10}\log d_i \right) = -\frac{1}{d_{k}}\\
        \\
        &\text{2nd. term; Shape Term}\\
        \\
        &\frac{\partial}{\partial d_{k}} \log f_{\mathbf{Z}}(\mathbf{z};\mathbf{g}) = \frac{\partial z_k}{\partial {d_k}}\cdot\frac{\partial}{\partial z_k}\cdot\log\:f_\mathbf{Z}(\mathbf{z};\mathbf{g})\\
        \\
        & \hspace{90pt}=-\frac{x_k}{d_k^2}\cdot\textcolor{red}{\frac{\partial}{\partial z_k}\cdot\log\:f_\mathbf{Z}(\mathbf{z};\mathbf{g})}\\
        \\
        \end{aligned}
        \nonumber
    \end{equation}
    
    \begin{equation}
        \begin{aligned}
        & 1st. term + 2nd. term\\
        \\
        & \mathbf{S}_{\mathbf{d}_k}(\mathbf{x};\mathbf{d})=-\frac{1}{d_k} + -\frac{x_k}{d_k^2}\cdot\textcolor{red}{\frac{\partial}{\partial z_k}\cdot\log\:f_\mathbf{Z}(\mathbf{z};\mathbf{g})}\\
        \\
        & \hspace{60pt} = -\frac{1}{d_k}\left(1+ -\frac{x_k}{d_k}\cdot\textcolor{red}{\frac{\partial}{\partial z_k}\cdot\log\:f_\mathbf{Z}(\mathbf{z};\mathbf{g})}\right)\\

        & \hspace{60pt} = -\frac{1}{d_k}\left(1+ -\frac{x_k}{d_k}\cdot\textcolor{red}{\mathbf{S}_{z_k}(\mathbf{z};\mathbf{g})}\right)
        \end{aligned}
        \nonumber
    \end{equation}

    \item shifting \\
    \begin{equation}
        \begin{aligned}
            \mathbf{S}_{d_k}(\mathbf{x};\mathbf{d})=-\mathbf{S}_{z_k}(\mathbf{z};\mathbf{g})
        \end{aligned}
        \nonumber
    \end{equation}
\end{itemize}

\section{SST Validation}

\subsection{Bivariate Gaussian Distribution}
\begin{itemize}
    \item
    Consider a Bivariate Gaussian distribution where the design variable $\mathrm{d}$ acts as a scaling parameter (Scale Family). The relationship between the physical variable $\mathrm{\mathbf{x}}$ and the normalized variable $\mathbf{z}$ is defined as follows:
    \begin{equation}
        \mathrm{\mathbf{x}} = \mathrm{diag}(\mathrm{\mathbf{d}}) \mathbf{z}, \quad \mathbf{z} \sim \mathcal{N}(\mathbf{1}, \boldsymbol{\Sigma}_0)
    \end{equation}
    where $\boldsymbol{\Sigma}_0 = \begin{bmatrix} 1 & \rho \\ \rho & 1 \end{bmatrix}$ is the constant correlation matrix.
\\

    \item Step 1: Direct Calculation of $\nabla_{\mathbf{z}} \log f_{\mathbf{Z}}(\mathbf{z})$
    The log-density of the normalized variable $\mathbf{z}$ is given by:
    \begin{equation}
        \log f_{\mathbf{Z}}(\mathbf{z}) = -\frac{1}{2} \log |2\pi \boldsymbol{\Sigma}_0| - \frac{1}{2}(\mathbf{z}-\mathbf{1})^\top \boldsymbol{\Sigma}_0^{-1} (\mathbf{z}-\mathbf{1})
    \end{equation}
    The score function with respect to $\mathbf{z}$ is obtained by direct differentiation:
    \begin{equation}
        \mathbf{S}_{\mathbf{z}}(\mathbf{z}) = \nabla_{\mathbf{z}} \log f_{\mathbf{Z}}(\mathbf{z}) = -\boldsymbol{\Sigma}_0^{-1}(\mathbf{z}-\mathbf{1})
    \end{equation}
\\

    \item Step 2: Direct Calculation of $\nabla_{\mathrm{\mathbf{d}}} \log f_{\mathrm{\mathbf{X}}}(\mathrm{\mathbf{x}}; \mathrm{\mathbf{d}})$
    The log-density in the physical space, considering the Jacobian $|\mathbf{J}| = \prod \mathrm{d}_i$, is:
    \begin{equation}
        \log f_{\mathrm{\mathbf{X}}}(\mathrm{\mathbf{x}}; \mathrm{\mathbf{d}}) = -\sum_{i=1}^{2} \log \mathrm{d}_i - \frac{1}{2}(\mathrm{\mathbf{x}}\cdot\mathrm{diag}^{-1}(\mathbf{d})-\mathbf{1})^\top \boldsymbol{\Sigma}_0^{-1} (\mathrm{\mathbf{x}}\cdot\mathrm{diag}^{-1}(\mathrm{\mathbf{d}})-\mathbf{1}) + C
    \end{equation}
    Differentiating with respect to $\mathrm{d}_k$ using the chain rule:
    \begin{itemize}
        \item \textbf{Jacobian Term:} $\frac{\partial}{\partial \mathrm{d}_k} (-\sum \log \mathrm{d}_i) = -\frac{1}{\mathrm{d}_k}$
        \item \textbf{Shape Term:} Let $\mathbf{u} = \mathrm{\mathbf{x}}\cdot\mathrm{diag}^{-1}(\mathrm{\mathbf{d}})$. Then $\frac{\partial \mathrm{u}_k}{\partial \mathrm{d}_k} = -\frac{\mathrm{x}_k}{\mathrm{d}_k^2}$.
        \begin{equation}
            \frac{\partial}{\partial \mathrm{d}_k} \text{Shape} = -\frac{1}{2} \cdot 2 \cdot \left[ \boldsymbol{\Sigma}_0^{-1}(\mathbf{u}-\mathbf{1}) \right]_k \cdot \left( -\frac{\mathrm{x}_k}{\mathrm{d}_k^2} \right) = \frac{\mathrm{x}_k}{\mathrm{d}_k^2} \left[ \boldsymbol{\Sigma}_0^{-1}(\mathbf{u}-\mathbf{1}) \right]_k
        \end{equation}
    \end{itemize}
    Total direct score $\mathrm{S}_{\mathrm{d}_k}$:
    \begin{equation}
        \mathrm{S}_{\mathrm{d}_k}^{\text{Direct}} = -\frac{1}{\mathrm{d}_k} + \frac{\mathrm{x}_k}{\mathrm{d}_k^2} \left[ \boldsymbol{\Sigma}_0^{-1}(\mathbf{u}-\mathbf{1}) \right]_k
    \end{equation}
\\

    \item Step 3: Verification via SST Theory
    According to SST theory for Scale Family, the transformation is defined as:
    \begin{equation}
        \mathrm{S}_{\mathrm{d}_k}^{\text{SST}} = -\frac{1}{\mathrm{d}_k} \left( 1 + \frac{\mathrm{x}_k}{\mathrm{d}_k} \mathrm{S}_{\mathbf{z}, k} \right)
    \end{equation}
    Substituting $\mathrm{S}_{\mathbf{z}, k} = \left[ -\boldsymbol{\Sigma}_0^{-1}(\mathbf{z}-\mathbf{1}) \right]_k$ from Step 1:
    \begin{equation}
        \mathrm{S}_{\mathrm{d}_k}^{\text{SST}} = -\frac{1}{\mathrm{d}_k} \left( 1 + \frac{\mathrm{x}_k}{\mathrm{d}_k} \left[ -\boldsymbol{\Sigma}_0^{-1}(\mathbf{u}-\mathbf{1}) \right]_k \right)
    \end{equation}
    Distributing terms:
    \begin{equation}
        \mathrm{S}_{\mathrm{d}_k}^{\text{SST}} = -\frac{1}{\mathrm{d}_k} + \frac{\mathrm{x}_k}{\mathrm{d}_k^2} \left[ \boldsymbol{\Sigma}_0^{-1}(\mathbf{u}-\mathbf{1}) \right]_k
    \end{equation}
    \textbf{Conclusion:} $\mathrm{S}_{\mathrm{d}_k}^{\text{Direct}} \equiv \mathrm{S}_{\mathrm{d}_k}^{\text{SST}}$. The mathematical and logical consistency of the SST theory is verified for correlated bivariate scaling cases.
\end{itemize}
